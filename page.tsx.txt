                      <div className="md:w-72 h-56 md:h-auto relative">
                        <img src={property.image} alt={property.title} className="w-full h-full object-cover" />
                        {property.verified && (
                          <span className="absolute top-3 left-3 bg-emerald-600 text-white text-xs font-bold px-2 py-0.5 rounded-md flex items-center space-x-1">
                            <CheckCircle2 className="w-3 h-3" />
                            <span>Verified</span>
                          </span>
                        )}
                      </div>

                      <div className="p-6 flex-1 flex flex-col justify-between">
                        <div>
                          <div className="flex items-center justify-between mb-2">
                            <div className="flex items-center text-slate-500 text-xs font-medium">
                              <MapPin className="w-3.5 h-3.5 text-indigo-600 mr-1" />
                              <span>{property.locality}, {property.city}</span>
                            </div>
                            <span className="text-lg font-black text-indigo-600">
                              ₹{property.price.toLocaleString('en-IN')} <span className="text-xs font-normal text-slate-500">/mo</span>
                            </span>
                          </div>

                          <h3 className="text-xl font-bold text-slate-900 mb-2">{property.title}</h3>
                          <p className="text-sm text-slate-600 mb-4 line-clamp-2">{property.description}</p>

                          <div className="flex flex-wrap gap-2 mb-4">
                            {property.amenities.map((a, i) => (
                              <span key={i} className="bg-slate-100 text-slate-700 text-xs font-medium px-3 py-1 rounded-lg">
                                {a}
                              </span>
                            ))}
                          </div>
                        </div>

                        <div className="pt-4 border-t border-slate-100 flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
                          <div className="flex items-center space-x-3">
                            <div className="bg-slate-100 p-2.5 rounded-full text-slate-700 font-bold text-xs">
                              {property.ownerName.charAt(0)}
                            </div>
                            <div>
                              <p className="text-[11px] text-slate-400 uppercase font-semibold">Direct Landlord</p>
                              <p className="text-xs font-bold text-slate-800">{property.ownerName} • <span className="text-indigo-600">{property.ownerContact}</span></p>
                            </div>
                          </div>

                          <div className="flex items-center space-x-2">
                            <button
                              onClick={() => {
                                if (!shortlisted.includes(property.id)) {
                                  setShortlisted([...shortlisted, property.id]);
                                  alert('Property added to your Shortlisted list!');
                                } else {
                                  alert('Property is already in your shortlist.');
                                }
                              }}
                              className="p-2.5 rounded-xl border border-slate-200 hover:bg-slate-50 text-slate-600 transition-colors"
                              title="Shortlist Property"
                            >
                              <Heart className={`w-4 h-4 ${shortlisted.includes(property.id) ? 'fill-rose-500 text-rose-500' : ''}`} />
                            </button>
                            <button
                              onClick={() => {
                                if (!user) {
                                  setAuthModalOpen(true);
                                  return;
                                }
                                if (!bookingRequests.includes(property.id)) {
                                  setBookingRequests([...bookingRequests, property.id]);
                                  alert('Booking inquiry sent successfully to the owner!');
                                } else {
                                  alert('Booking inquiry already active for this property.');
                                }
                              }}
                              className="bg-indigo-600 hover:bg-indigo-700 text-white px-5 py-2.5 rounded-xl text-xs font-bold transition-all shadow-md shadow-indigo-200"
                            >
                              Contact Owner / Book
                            </button>
                          </div>
                        </div>
                      </div>
                    </div>
                  ))
                )}
              </div>
            </div>
          </div>
        )}

        {/* C & D. DASHBOARDS */}
        {activeTab === 'dashboard' && user && (
          <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-10">
            
            {/* TENANT DASHBOARD */}
            {user.role === 'tenant' && (
              <div className="space-y-8">
                <div>
                  <h1 className="text-3xl font-black text-slate-900">Tenant Dashboard</h1>
                  <p className="text-slate-500 text-sm mt-1">Manage your shortlisted spaces, active inquiries, and rental agreements.</p>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                  <div className="bg-white p-6 rounded-2xl border border-slate-200 shadow-sm">
                    <div className="flex items-center justify-between mb-4">
                      <span className="text-xs font-bold text-slate-400 uppercase">Shortlisted Properties</span>
                      <Heart className="w-5 h-5 text-rose-500" />
                    </div>
                    <p className="text-3xl font-black text-slate-900">{shortlisted.length}</p>
                  </div>
                  <div className="bg-white p-6 rounded-2xl border border-slate-200 shadow-sm">
                    <div className="flex items-center justify-between mb-4">
                      <span className="text-xs font-bold text-slate-400 uppercase">Active Inquiries / Bookings</span>
                      <Bookmark className="w-5 h-5 text-indigo-600" />
                    </div>
                    <p className="text-3xl font-black text-slate-900">{bookingRequests.length}</p>
                  </div>
                  <div className="bg-white p-6 rounded-2xl border border-slate-200 shadow-sm">
                    <div className="flex items-center justify-between mb-4">
                      <span className="text-xs font-bold text-slate-400 uppercase">Rental Status</span>
                      <ShieldCheck className="w-5 h-5 text-emerald-600" />
                    </div>
                    <p className="text-lg font-bold text-emerald-600">Verified Tenant Profile</p>
                  </div>
                </div>

                <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm">
                  <h3 className="text-lg font-bold text-slate-900 mb-4">Your Shortlisted Properties</h3>
                  {shortlisted.length === 0 ? (
                    <p className="text-sm text-slate-500">No properties shortlisted yet.</p>
                  ) : (
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      {properties.filter(p => shortlisted.includes(p.id)).map(p => (
                        <div key={p.id} className="p-4 rounded-xl border border-slate-200 flex items-center space-x-4">
                          <img src={p.image} alt={p.title} className="w-20 h-20 rounded-lg object-cover" />
                          <div className="flex-1">
                            <h4 className="font-bold text-slate-900 text-sm">{p.title}</h4>
                            <p className="text-xs text-slate-500">{p.locality}, {p.city}</p>
                            <p className="text-xs font-bold text-indigo-600 mt-1">₹{p.price.toLocaleString('en-IN')} /mo</p>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </div>
            )}

            {/* LANDLORD PORTAL & ROOM MANAGEMENT */}
            {user.role === 'landlord' && (
              <div className="space-y-8">
                <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
                  <div>
                    <h1 className="text-3xl font-black text-slate-900">Landlord Property & Room Management</h1>
                    <p className="text-slate-500 text-sm mt-1">Monitor room occupancy status and onboard tenants securely.</p>
                  </div>
                  <button 
                    onClick={() => alert('Add Property modal triggered.')}
                    className="bg-indigo-600 hover:bg-indigo-700 text-white px-5 py-2.5 rounded-xl text-xs font-bold transition-all shadow-md shadow-indigo-200 flex items-center space-x-2 w-fit"
                  >
                    <Plus className="w-4 h-4" />
                    <span>Add New Property</span>
                  </button>
                </div>

                <div className="space-y-8">
                  {properties.map(property => (
                    <div key={property.id} className="bg-white rounded-2xl border border-slate-200 overflow-hidden shadow-sm">
                      <div className="p-6 bg-slate-50 border-b border-slate-200 flex flex-col md:flex-row md:items-center justify-between gap-4">
                        <div className="flex items-center space-x-4">
                          <img src={property.image} alt={property.title} className="w-16 h-16 rounded-xl object-cover" />
                          <div>
                            <h3 className="text-lg font-bold text-slate-900">{property.title}</h3>
                            <p className="text-xs text-slate-500">{property.locality}, {property.city} • ₹{property.price.toLocaleString('en-IN')} /mo</p>
                          </div>
                        </div>
                        <div className="flex items-center space-x-2">
                          <span className="bg-indigo-100 text-indigo-700 text-xs font-bold px-3 py-1.5 rounded-xl">
                            {property.rooms.filter(r => r.status === 'Empty').length} Rooms Available
                          </span>
                        </div>
                      </div>

                      <div className="p-6">
                        <h4 className="text-xs font-bold text-slate-400 uppercase tracking-wider mb-4">Room Status Grid</h4>
                        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                          {property.rooms.map(room => (
                            <div 
                              key={room.id} 
                              className={`p-5 rounded-2xl border transition-all ${room.status === 'Empty' ? 'bg-emerald-50/30 border-emerald-200' : 'bg-rose-50/30 border-rose-200'}`}
                            >
                              <div className="flex items-center justify-between mb-3">
                                <span className="font-bold text-slate-900 text-sm">Room {room.roomNumber} ({room.type})</span>
                                <span className={`text-xs font-bold px-2.5 py-1 rounded-full ${room.status === 'Empty' ? 'bg-emerald-100 text-emerald-700' : 'bg-rose-100 text-rose-700'}`}>
                                  {room.status === 'Empty' ? 'Empty (Available)' : 'Full (Occupied)'}
                                </span>
                              </div>

                              {room.status === 'Full' && room.tenant ? (
                                <div className="space-y-2 mt-3 pt-3 border-t border-slate-200 text-xs">
                                  <p className="font-bold text-slate-800">{room.tenant.name}</p>
                                  <p className="text-slate-600">Ph: {room.tenant.mobile}</p>
                                  <p className="text-slate-600">Aadhaar: {room.tenant.aadhaar}</p>
                                  <p className="text-slate-600">Rent: ₹{room.tenant.monthlyRent}/mo (Due: {room.tenant.dueDate})</p>
                                  <button
                                    onClick={() => handleVacateRoom(property.id, room.id)}
                                    className="mt-2 w-full bg-rose-600 hover:bg-rose-700 text-white font-bold py-1.5 rounded-lg transition-colors"
                                  >
                                    Vacate Room
                                  </button>
                                </div>
                              ) : (
                                <div className="mt-4">
                                  <button
                                    onClick={() => setOnboardingModal({ propertyId: property.id, roomId: room.id })}
                                    className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-2 rounded-xl text-xs transition-all shadow-sm flex items-center justify-center space-x-1.5"
                                  >
                                    <UserPlus className="w-3.5 h-3.5" />
                                    <span>Onboard Tenant</span>
                                  </button>
                                </div>
                              )}
                            </div>
                          ))}
                        </div>
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            )}

          </div>
        )}

      </main>

      {/* TENANT ONBOARDING MODAL */}
      {onboardingModal && (
        <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl max-w-lg w-full p-8 shadow-2xl border border-slate-100 relative">
            <button
              onClick={() => setOnboardingModal(null)}
              className="absolute top-6 right-6 text-slate-400 hover:text-slate-700"
            >
              <X className="w-6 h-6" />
            </button>

            <h3 className="text-2xl font-black text-slate-900 mb-1">Tenant Onboarding</h3>
            <p className="text-sm text-slate-500 mb-6">Enter tenant details and rent agreement terms for room check-in.</p>

            <form onSubmit={handleOnboardSubmit} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Full Name</label>
                <input
                  type="text"
                  required
                  placeholder="e.g. Rahul Verma"
                  value={tenantForm.name}
                  onChange={(e) => setTenantForm({ ...tenantForm, name: e.target.value })}
                  className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Mobile Number</label>
                  <input
                    type="tel"
                    required
                    maxLength={10}
                    placeholder="10-digit number"
                    value={tenantForm.mobile}
                    onChange={(e) => setTenantForm({ ...tenantForm, mobile: e.target.value })}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Aadhaar Number</label>
                  <input
                    type="text"
                    required
                    placeholder="12-digit Aadhaar"
                    value={tenantForm.aadhaar}
                    onChange={(e) => {
                      setTenantForm({ ...tenantForm, aadhaar: e.target.value });
                      validateAadhaar(e.target.value);
                    }}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                  />
                  {aadhaarError && <p className="text-[10px] text-rose-600 font-bold mt-1">{aadhaarError}</p>}
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Permanent Address</label>
                <input
                  type="text"
                  required
                  placeholder="City, State"
                  value={tenantForm.address}
                  onChange={(e) => setTenantForm({ ...tenantForm, address: e.target.value })}
                  className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                />
              </div>

              <div className="grid grid-cols-3 gap-3">
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 uppercase mb-1">Monthly Rent (₹)</label>
                  <input
                    type="number"
                    required
                    value={tenantForm.monthlyRent}
                    onChange={(e) => setTenantForm({ ...tenantForm, monthlyRent: Number(e.target.value) })}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                  />
                </div>
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 uppercase mb-1">Deposit (₹)</label>
                  <input
                    type="number"
                    required
                    value={tenantForm.securityDeposit}
                    onChange={(e) => setTenantForm({ ...tenantForm, securityDeposit: Number(e.target.value) })}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                  />
                </div>
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 uppercase mb-1">Due Date</label>
                  <input
                    type="text"
                    required
                    placeholder="e.g. 5th of month"
                    value={tenantForm.dueDate}
                    onChange={(e) => setTenantForm({ ...tenantForm, dueDate: e.target.value })}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2.5 text-sm focus:outline-none focus:border-indigo-500"
                  />
                </div>
              </div>

              <div className="pt-4">
                <button
                  type="submit"
                  className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-3 rounded-xl shadow-md shadow-indigo-200 transition-all text-sm"
                >
                  Save & Onboard Tenant
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* AUTH MODAL */}
      {authModalOpen && (
        <div className="fixed inset-0 z-50 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl max-w-md w-full p-8 shadow-2xl border border-slate-100 relative">
            <button
              onClick={() => { setAuthModalOpen(false); setOtpSent(false); }}
              className="absolute top-6 right-6 text-slate-400 hover:text-slate-700"
            >
              <X className="w-6 h-6" />
            </button>

            <div className="text-center mb-6">
              <div className="bg-indigo-50 text-indigo-600 w-12 h-12 rounded-2xl flex items-center justify-center mx-auto mb-3">
                <Building2 className="w-6 h-6" />
              </div>
              <h3 className="text-2xl font-black text-slate-900">Welcome to NestFinder</h3>
              <p className="text-sm text-slate-500 mt-1">Sign in with your mobile number to continue</p>
            </div>

            {!otpSent ? (
              <div className="space-y-4">
                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-2">Select Your Role</label>
                  <div className="grid grid-cols-2 gap-3">
                    <button
                      type="button"
                      onClick={() => setSelectedRole('tenant')}
                      className={`py-3 rounded-xl font-bold text-xs border transition-all ${selectedRole === 'tenant' ? 'bg-indigo-600 text-white border-indigo-600 shadow-md shadow-indigo-200' : 'bg-slate-50 text-slate-700 border-slate-200'}`}
                    >
                      Tenant
                    </button>
                    <button
                      type="button"
                      onClick={() => setSelectedRole('landlord')}
                      className={`py-3 rounded-xl font-bold text-xs border transition-all ${selectedRole === 'landlord' ? 'bg-indigo-600 text-white border-indigo-600 shadow-md shadow-indigo-200' : 'bg-slate-50 text-slate-700 border-slate-200'}`}
                    >
                      Landlord / Owner
                    </button>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Mobile Number</label>
                  <div className="flex items-center bg-slate-50 border border-slate-200 rounded-xl px-3 py-2.5">
                    <span className="text-slate-400 font-bold text-sm mr-2">+91</span>
                    <input
                      type="tel"
                      maxLength={10}
                      placeholder="Enter 10-digit mobile number"
                      value={mobileInput}
                      onChange={(e) => setMobileInput(e.target.value)}
                      className="w-full bg-transparent text-sm font-semibold focus:outline-none"
                    />
                  </div>
                </div>

                <button
                  onClick={handleSendOtp}
                  className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-3 rounded-xl shadow-md shadow-indigo-200 transition-all text-sm"
                >
                  Send OTP
                </button>
              </div>
            ) : (
              <div className="space-y-4">
                <div className="bg-indigo-50 border border-indigo-100 p-4 rounded-xl text-center">
                  <p className="text-xs text-indigo-700 font-medium mb-1">Mock OTP Sent to +91 {mobileInput}</p>
                  <p className="text-lg font-black text-indigo-900 tracking-widest">{mockOtpCode}</p>
                </div>

                <div>
                  <label className="block text-xs font-bold text-slate-700 uppercase mb-1">Enter 6-Digit OTP</label>
                  <input
                    type="text"
                    maxLength={6}
                    placeholder="Enter OTP (or 123456)"
                    value={otpInput}
                    onChange={(e) => setOtpInput(e.target.value)}
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-center text-lg font-bold tracking-widest focus:outline-none focus:border-indigo-500"
                  />
                </div>

                <button
                  onClick={handleVerifyOtp}
                  className="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-bold py-3 rounded-xl shadow-md shadow-emerald-200 transition-all text-sm"
                >
                  Verify & Proceed
                </button>
              </div>
            )}
          </div>
        </div>
      )}

      {/* FOOTER */}
      <footer className="bg-white border-t border-slate-200 py-8 mt-auto">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 flex flex-col sm:flex-row items-center justify-between text-xs text-slate-500">
          <p>© 2026 NestFinder. Zero Brokerage Direct Living Spaces in India.</p>
          <div className="flex space-x-6 mt-4 sm:mt-0 font-medium">
            <a href="#" className="hover:text-indigo-600">Privacy Policy</a>
            <a href="#" className="hover:text-indigo-600">Terms of Service</a>
            <a href="#" className="hover:text-indigo-600">Support</a>
          </div>
        </div>
      </footer>

    </div>
  );
}
